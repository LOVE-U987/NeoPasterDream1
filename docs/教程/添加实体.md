# 添加实体

> 本教程将指导你从零开始添加一个自定义实体。

---

## 概述

我们将添加一个名为 `dream_spirit` 的实体,具有以下特性:
- 中立生物
- 有自定义模型和动画
- 有 AI 行为
- 有掉落物

---

## 步骤 1: 创建实体类

### 1.1 创建实体类文件

创建文件 `PasterDream/src/main/java/com/pasterdream/pasterdreammod/entity/mob/DreamSpiritEntity.java`:

```java
package com.pasterdream.pasterdreammod.entity.mob;

import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.MobCategory;
import net.minecraft.world.entity.PathfinderMob;
import net.minecraft.world.entity.ai.attributes.AttributeSupplier;
import net.minecraft.world.entity.ai.attributes.Attributes;
import net.minecraft.world.entity.ai.goal.FloatGoal;
import net.minecraft.world.entity.ai.goal.MeleeAttackGoal;
import net.minecraft.world.entity.ai.goal.RandomLookAroundGoal;
import net.minecraft.world.entity.ai.goal.WaterAvoidingRandomStrollGoal;
import net.minecraft.world.entity.ai.goal.target.NearestAttackableTargetGoal;
import net.minecraft.world.entity.monster.Monster;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.level.Level;

/**
 * 梦境精灵实体
 * 
 * 一种中立的精灵生物,会在染梦维度生成。
 * 玩家攻击后会变为敌对。
 */
public class DreamSpiritEntity extends Monster {
    public DreamSpiritEntity(EntityType<? extends Monster> entityType, Level level) {
        super(entityType, level);
    }

    /**
     * 创建实体属性
     * 
     * @return 属性构建器
     */
    public static AttributeSupplier.Builder createAttributes() {
        return Monster.createMonsterAttributes()
            .add(Attributes.MAX_HEALTH, 30.0)
            .add(Attributes.MOVEMENT_SPEED, 0.3)
            .add(Attributes.ATTACK_DAMAGE, 8.0)
            .add(Attributes.FOLLOW_RANGE, 32.0);
    }

    @Override
    protected void registerGoals() {
        this.goalSelector.addGoal(0, new FloatGoal(this));
        this.goalSelector.addGoal(1, new MeleeAttackGoal(this, 1.2, false));
        this.goalSelector.addGoal(2, new WaterAvoidingRandomStrollGoal(this, 0.8));
        this.goalSelector.addGoal(3, new RandomLookAroundGoal(this));
        
        this.targetSelector.addGoal(1, new NearestAttackableTargetGoal<>(this, Player.class, true));
    }
}
```

## 步骤 2: 注册实体

### 2.1 在 PDEntities.java 中注册

打开 `PasterDream/src/main/java/com/pasterdream/pasterdreammod/registry/PDEntities.java`,添加注册:

```java
// 在 PDEntities 类中添加
public static final DeferredHolder<EntityType<DreamSpiritEntity>, EntityType<DreamSpiritEntity>> DREAM_SPIRIT = 
    ENTITY_TYPES.register("dream_spirit", () -> EntityType.Builder.of(DreamSpiritEntity::new, MobCategory.MONSTER)
        .sized(0.8f, 1.6f)
        .clientTrackingRange(8)
        .build("dream_spirit"));
```

### 2.2 注册属性

在 `PDEntityEvents.java` 中注册属性:

```java
@SubscribeEvent
public static void onEntityAttributeCreation(EntityAttributeCreationEvent event) {
    event.put(PDEntities.DREAM_SPIRIT.get(), DreamSpiritEntity.createAttributes().build());
}
```

### 2.3 注册刷怪蛋

在 `PDItemsSpawnEggs.java` 中添加:

```java
public static final DeferredItem<Item> DREAM_SPIRIT_SPAWN_EGG = 
    PDItems.ITEMS.register("dream_spirit_spawn_egg", () -> new ForgeSpawnEggItem(
        PDEntities.DREAM_SPIRIT,
        0x9B59B6,  // 主色
        0x8E44AD,  // 次色
        new Item.Properties()));
```

## 步骤 3: 创建渲染器

### 3.1 创建渲染器类

创建文件 `PasterDream/src/main/java/com/pasterdream/pasterdreammod/client/renderer/entity/DreamSpiritRenderer.java`:

```java
package com.pasterdream.pasterdreammod.client.renderer.entity;

import com.pasterdream.pasterdreammod.client.model.entity.DreamSpiritModel;
import com.pasterdream.pasterdreammod.entity.mob.DreamSpiritEntity;
import com.pasterdream.pasterdreammod.PasterDreamMod;
import net.minecraft.client.renderer.entity.EntityRendererProvider;
import net.minecraft.client.renderer.entity.MobRenderer;
import net.minecraft.resources.ResourceLocation;

/**
 * 梦境精灵渲染器
 */
public class DreamSpiritRenderer extends MobRenderer<DreamSpiritEntity, DreamSpiritModel<DreamSpiritEntity>> {
    private static final ResourceLocation TEXTURE = 
        ResourceLocation.fromNamespaceAndPath(PasterDreamMod.MOD_ID, "textures/entity/dream_spirit.png");

    public DreamSpiritRenderer(EntityRendererProvider.Context context) {
        super(context, new DreamSpiritModel<>(context.bakeLayer(DreamSpiritModel.LAYER)), 0.5f);
    }

    @Override
    public ResourceLocation getTextureLocation(DreamSpiritEntity entity) {
        return TEXTURE;
    }
}
```

### 3.2 创建模型类

创建文件 `PasterDream/src/main/java/com/pasterdream/pasterdreammod/client/model/entity/DreamSpiritModel.java`:

```java
package com.pasterdream.pasterdreammod.client.model.entity;

import com.mojang.blaze3d.vertex.PoseStack;
import com.mojang.blaze3d.vertex.VertexConsumer;
import com.pasterdream.pasterdreammod.PasterDreamMod;
import net.minecraft.client.model.EntityModel;
import net.minecraft.client.model.geom.ModelLayerLocation;
import net.minecraft.client.model.geom.ModelPart;
import net.minecraft.client.model.geom.PartPose;
import net.minecraft.client.model.geom.builders.*;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.world.entity.Entity;

/**
 * 梦境精灵模型
 */
public class DreamSpiritModel<T extends Entity> extends EntityModel<T> {
    public static final ModelLayerLocation LAYER = 
        new ModelLayerLocation(ResourceLocation.fromNamespaceAndPath(PasterDreamMod.MOD_ID, "dream_spirit"), "main");

    private final ModelPart body;
    private final ModelPart head;

    public DreamSpiritModel(ModelPart root) {
        this.body = root.getChild("body");
        this.head = root.getChild("head");
    }

    public static LayerDefinition createBodyLayer() {
        MeshDefinition meshdefinition = new MeshDefinition();
        PartDefinition partdefinition = meshdefinition.getRoot();

        PartDefinition body = partdefinition.addOrReplaceChild("body", CubeListBuilder.create()
            .texOffs(0, 16)
            .addBox(-3.0f, 0.0f, -2.0f, 6.0f, 8.0f, 4.0f, new CubeDeformation(0.0f)),
            PartPose.offset(0.0f, 12.0f, 0.0f));

        PartDefinition head = partdefinition.addOrReplaceChild("head", CubeListBuilder.create()
            .texOffs(0, 0)
            .addBox(-4.0f, -8.0f, -4.0f, 8.0f, 8.0f, 8.0f, new CubeDeformation(0.0f)),
            PartPose.offset(0.0f, 12.0f, 0.0f));

        return LayerDefinition.create(meshdefinition, 64, 64);
    }

    @Override
    public void setupAnim(T entity, float limbSwing, float limbSwingAmount, float ageInTicks, float netHeadYaw, float headPitch) {
        this.head.yRot = netHeadYaw * ((float) Math.PI / 180.0f);
        this.head.xRot = headPitch * ((float) Math.PI / 180.0f);
    }

    @Override
    public void renderToBuffer(PoseStack poseStack, VertexConsumer vertexConsumer, int packedLight, int packedOverlay, int color) {
        body.render(poseStack, vertexConsumer, packedLight, packedOverlay, color);
        head.render(poseStack, vertexConsumer, packedLight, packedOverlay, color);
    }
}
```

## 步骤 4: 注册渲染器

### 4.1 注册模型层

在 `ClientSetup.java` 中注册:

```java
@SubscribeEvent
public static void onEntityRenderers(EntityRenderersEvent.RegisterLayerDefinitions event) {
    event.registerLayerDefinition(DreamSpiritModel.LAYER, DreamSpiritModel::createBodyLayer);
}
```

### 4.2 注册渲染器

在同一文件中注册:

```java
@SubscribeEvent
public static void onEntityRenderers(EntityRenderersEvent.RegisterRenderers event) {
    event.registerEntityRenderer(PDEntities.DREAM_SPIRIT.get(), DreamSpiritRenderer::new);
}
```

## 步骤 5: 创建资源文件

### 5.1 纹理

创建纹理文件 `PasterDream/src/main/resources/assets/pasterdream/textures/entity/dream_spirit.png`。

**注意**: 纹理需要是 64x64 像素的 PNG 文件 (根据模型 UV 布局)。

### 5.2 语言文件

在 `en_us.json` 中添加:

```json
{
    "entity.pasterdream.dream_spirit": "Dream Spirit",
    "item.pasterdream.dream_spirit_spawn_egg": "Dream Spirit Spawn Egg"
}
```

在 `zh_cn.json` 中添加:

```json
{
    "entity.pasterdream.dream_spirit": "梦境精灵",
    "item.pasterdream.dream_spirit_spawn_egg": "梦境精灵刷怪蛋"
}
```

### 5.3 掉落物表

创建文件 `PasterDream/src/main/resources/data/pasterdream/loot_table/entities/dream_spirit.json`:

```json
{
    "type": "minecraft:entity",
    "pools": [
        {
            "rolls": 1.0,
            "entries": [
                {
                    "type": "minecraft:item",
                    "name": "pasterdream:dream_shard"
                }
            ],
            "conditions": [
                {
                    "condition": "minecraft:killed_by_player"
                }
            ]
        }
    ]
}
```
